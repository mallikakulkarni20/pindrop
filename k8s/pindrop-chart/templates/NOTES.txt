2. For local Minikube testing (macOS), use port-forwarding:

   # Forward frontend
   kubectl port-forward svc/{{ include "pindrop.fullname" . }}-frontend 8080:80 -n {{ include "pindrop.namespace" . }} &

   # Forward backend
   kubectl port-forward svc/{{ include "pindrop.fullname" . }}-backend 3001:3001 -n {{ include "pindrop.namespace" . }} &

   # Access at:
   http://localhost:8080

-----------------------------------------------------------------------------
  USEFUL COMMANDS
-----------------------------------------------------------------------------

# Check pod status
kubectl get pods -n {{ include "pindrop.namespace" . }}

# View logs
kubectl logs -l app.kubernetes.io/component=backend -n {{ include "pindrop.namespace" . }}
kubectl logs -l app.kubernetes.io/component=frontend -n {{ include "pindrop.namespace" . }}

# Upgrade with new values
helm upgrade {{ .Release.Name }} ./pindrop-chart -f values-prod.yaml

# Rollback to previous version
helm rollback {{ .Release.Name }} 1

# Uninstall
helm uninstall {{ .Release.Name }} -n {{ include "pindrop.namespace" . }}
